```python
# ---------------  Hydrate.py  ---------------
# AtCoder の問題「Hydrate」の解答例（コメント付き）

# ① 入力を取得
blueball_a, blueball_b, redball_c, time_d = map(int, input().split())

# ② 1 回の追加で青玉が増える速さ
#     1 秒あたりに赤玉が減る速さ * 時間
#     （つまり 1 回の追加で赤玉が減る個数）
red_per_round = redball_c * time_d

# ③ 青玉が増える速度が赤玉が減る速度を上回る／等しい場合
#     → 追いつかないので答えは -1
if blueball_b >= red_per_round:
    print(-1)
else:
    # ④ 追いつくのに必要な最小回数を計算
    #     n * (red_per_round - blueball_b) >= blueball_a
    #     → n = ceil( blueball_a / (red_per_round - blueball_b) )
    denom = red_per_round - blueball_b  # 正の整数
    n = (blueball_a + denom - 1) // denom  # 切り上げ除算
    print(n)
```

```python
# ---------------  Play_Snuke.py  ---------------
# 店の数を入力
shop_count = int(input())

# 購入できる商品の最安値
# None は「まだ購入できる店が見つかっていない」ことを表す
minimum_price = None

for _ in range(shop_count):
    # arrival_time: 店に到着するまでの時間
    # price: 商品の価格
    # sold_out_time: 商品が売り切れるまでの時間
    arrival_time, price, sold_out_time = map(int, input().split())

    # 到着時刻が売り切れる時刻より前なら購入できる
    if arrival_time < sold_out_time:
        # 最初に見つけた商品、または現在の最安値より安い商品なら更新
        if minimum_price is None or price < minimum_price:
            minimum_price = price

# 購入できる店がなければ -1
if minimum_price is None:
    print(-1)
else:
    print(minimum_price)
```
