```python
# ---------------  Shiritori.py  ---------------
# 入力される単語の個数
n = int(input())

# n 個の単語をリストに保存する
words = []
for _ in range(n):
    words.append(input())

# 最初は「しりとりが成立している」と仮定する
is_shiritori = True

# set は同じ値を 1 つにまとめる。
# 要素数が違えば、同じ単語が 2 回以上登場している。
if len(words) != len(set(words)):
    is_shiritori = False

# 前の単語の最後の文字と、次の単語の最初の文字を順に比べる
for i in range(n - 1):
    previous_word = words[i]
    next_word = words[i + 1]

    # つながっていなければ、しりとりは成立しない
    if previous_word[-1] != next_word[0]:
        is_shiritori = False
        break

# 全ての条件を満たしていれば Yes、そうでなければ No を出力する
if is_shiritori:
    print("Yes")
else:
    print("No")
```

```python
# -------------- Billiards.py -------------
# 出発点 S = (Sx, Sy)、壁で反射する前の目標点 G = (Gx, Gy) を受け取る
Sx, Sy, Gx, Gy = map(int, input().split())

# 反射を考える代わりに、G を x 軸について反転した点 (Gx, -Gy) へ
# まっすぐ進むと考える。この直線と x 軸の交点が答えになる。
#
# S から反転後の G までの縦方向の距離は Sy + Gy。
# x 軸に着くまでに下がる距離は Sy なので、直線全体のうち
# Sy / (Sy + Gy) だけ進んだ位置に交点がある。
ratio = Sy / (Sy + Gy)

# x 座標も同じ割合だけ Sx から Gx へ進めばよい。
answer_x = Sx + (Gx - Sx) * ratio

print(answer_x)

# この解法は計算量 O(1)、追加メモリ O(1)。
# 点を反転して直線との交点を求める方法が、シミュレーションをせずに済む最も簡潔な解き方。
```
